   480→   480→    suspend fun getInstalledApps(): List<AppInfo> = withContext(Dispatchers.IO) {
   481→   481→        val pm = context.packageManager
   482→   482→        val intent = Intent(Intent.ACTION_MAIN, null).apply {
   483→   483→            addCategory(Intent.CATEGORY_LAUNCHER)
   484→   484→        }
   485→   485→        
   486→   486→        val resolveInfos: List<ResolveInfo> = pm.queryIntentActivities(intent, 0)
   487→   487→        
   488→   488→        resolveInfos.mapNotNull { resolveInfo ->
   489→   489→            try {
   490→   490→                val packageName = resolveInfo.activityInfo.packageName
   491→   491→                // Skip our own app
   492→   492→                if (packageName == context.packageName) return@mapNotNull null
   493→   493→                
   494→   494→                val name = resolveInfo.loadLabel(pm).toString()
   495→   495→                val icon = resolveInfo.loadIcon(pm)
   496→   496→                
   497→   497→                val isEmulator = emulatorKeywords.any { 
   498→   498→                    packageName.lowercase().contains(it) || name.lowercase().contains(it)
   499→   499→                }
   500→   500→                
   501→   501→                AppInfo(
   502→   502→                    name = name,
   503→   503→                    packageName = packageName,
   504→   504→                    icon = icon,
   505→   505→                    isEmulator = isEmulator
   506→   506→                )
   507→   507→            } catch (e: Exception) {
   508→   508→                null
   509→   509→            }
   510→   510→        }.sortedBy { it.name }
   511→   511→    }

   519→   519→            }
   520→   520→        } catch (e: Exception) {
   521→   521→            e.printStackTrace()
   522→   522→        }
   523→   523→    }
