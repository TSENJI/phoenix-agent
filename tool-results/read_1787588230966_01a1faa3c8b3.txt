   461→    val name: String,
   462→    val packageName: String,
   463→    val icon: Drawable,
   464→    val isEmulator: Boolean = false
   465→)
   466→
   467→class AppScanner(private val context: Context) {
   468→    
   469→    // Known emulator package keywords for categorization
   470→    private val emulatorKeywords = listOf(
   471→        "retroarch", "emulator", "drastic", "citra", "azahar", "dolphin", 
   472→        "ppsspp", "yuzu", "suyu", "sudachi", "skyline", "eden", "aethersx2",
   473→        "nethersx2", "duckstation", "mupen64", "m64plus", "fzurita", "lime3ds",
   474→        "mandarine", "pizzaboy", "myboy", "myoldboy", "snes9x", "superretro",
   475→        "flycast", "redream", "vita3k", "melonds", "yaba", "epsxe", "fpse",
   476→        "lemuroid", "classicboy", "nes.emu", "gba.emu", "gbc.emu", "md.emu",
   477→        "snes9xex", "mame4droid", "neo.emu", "pce.emu"
   478→    )
   479→
   480→    suspend fun getInstalledApps(): List<AppInfo> = withContext(Dispatchers.IO) {
   481→        val pm = context.packageManager
   482→        val intent = Intent(Intent.ACTION_MAIN, null).apply {
   483→            addCategory(Intent.CATEGORY_LAUNCHER)
   484→        }
   485→        
   486→        val resolveInfos: List<ResolveInfo> = pm.queryIntentActivities(intent, 0)
   487→        
   488→        resolveInfos.mapNotNull { resolveInfo ->
   489→            try {
   490→                val packageName = resolveInfo.activityInfo.packageName
   491→                // Skip our own app
   492→                if (packageName == context.packageName) return@mapNotNull null
   493→                
   494→                val name = resolveInfo.loadLabel(pm).toString()
   495→                val icon = resolveInfo.loadIcon(pm)
   496→                
   497→                val isEmulator = emulatorKeywords.any { 
   498→                    packageName.lowercase().contains(it) || name.lowercase().contains(it)
   499→                }
   500→                
   501→                AppInfo(
   502→                    name = name,
   503→                    packageName = packageName,
   504→                    icon = icon,
   505→                    isEmulator = isEmulator
   506→                )
   507→            } catch (e: Exception) {
   508→                null
   509→            }
   510→        }.sortedBy { it.name }
   511→    }
   512→    
   513→    fun launchApp(packageName: String) {
   514→        try {
   515→            val launchIntent = context.packageManager.getLaunchIntentForPackage(packageName)
   516→            if (launchIntent != null) {
   517→                launchIntent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
   518→                context.startActivity(launchIntent)
   519→            }
   520→        } catch (e: Exception) {
   521→            e.printStackTrace()
   522→        }
   523→    }
   524→}
